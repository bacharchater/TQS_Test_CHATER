  \subsection{Énumérations}
\begin{lstlisting}[style=stcode]
{attribute 'qualified_only'}
{attribute 'strict'}
{attribute 'to_string'}
TYPE E_ModeCarrousel :
(
    INIT   := 0,
    AUTO   := 1,
    DEFAUT := 2
);
END_TYPE

{attribute 'qualified_only'}
{attribute 'strict'}
{attribute 'to_string'}
TYPE E_EtatCycle :
(
    ATTENTE       := 0,
    COMPTAGE      := 1,
    BAC_PLEIN     := 2,
    VERIN_SORTIE  := 3,
    VERIN_RENTREE := 4,
    BAC_SUIVANT   := 5,
    ATTENTE_BAC   := 6
);
END_TYPE
\end{lstlisting}

  \begin{notebox}{Lecture en ligne}
    Les énumérations rendent le mode et l'état courant directement compréhensibles dans TwinCAT, sans devoir interpréter des numéros d'étape.
  \end{notebox}

  \subsection{Interface de \texttt{FB\_Carrousel}}
\begin{lstlisting}[style=stcode]
FUNCTION_BLOCK FB_Carrousel
VAR_INPUT
    bLaser    : BOOL; // detection piece
    bZero     : BOOL; // origine carrousel
    nPieceBac : INT;  // consigne / bac
    tVerin    : TIME; // temps mouvement
    aResetBac : ARRAY[1..8] OF BOOL;
END_VAR

VAR_OUTPUT
    bVerinAv : BOOL; // cmd avant
    bVerinAr : BOOL; // cmd arriere
    nBac     : INT;  // bac en position
    bDefaut  : BOOL; // defaut carrousel
    eModeHmi      : E_ModeCarrousel;
    eEtatCycleHmi : E_EtatCycle;
    aPieceBacHmi  : ARRAY[1..8] OF INT;
    nImpHmi       : INT;
END_VAR

VAR
    eMode      : E_ModeCarrousel
                 := E_ModeCarrousel.INIT;
    eEtatCycle : E_EtatCycle
                 := E_EtatCycle.ATTENTE;
    aPieceBac  : ARRAY[1..8] OF INT;
    nImp       : INT := 0;
    fbLaser      : R_TRIG;
    fbTempoVerin : TON;
    aTempoReset  : ARRAY[1..8] OF TON;
    bResetGlobal : BOOL;
    i            : INT;
    j            : INT;
END_VAR
\end{lstlisting}

  \begin{primarycard}
    \textbf{Variables exposées à la Visualization}\par\small
    \texttt{MAIN} relie les quatre sorties HMI du bloc à \texttt{eModeOut}, \texttt{eEtatCycleOut}, \texttt{aPieceBacOut} et \texttt{nImpOut}. Il dérive aussi \texttt{sModeVisu}, \texttt{sEtatCycleVisu} et \texttt{aBacActifVisu[1..8]} pour l'affichage.
  \end{primarycard}
